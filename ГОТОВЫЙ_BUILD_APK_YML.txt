name: Build Gevork Billiardinsky v4.1 APK

on:
  push:
    branches: [main]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      - name: Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - name: Android SDK packages
        shell: bash
        run: |
          SDKMANAGER="$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager"
          if [ ! -x "$SDKMANAGER" ]; then
            echo "Android SDK command-line tools not found at $SDKMANAGER"
            exit 1
          fi
          "$SDKMANAGER" --install "platforms;android-35" "build-tools;35.0.0"
      - name: Gradle 8.10.2
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.10.2'
      - name: Check JavaScript
        run: node --check app/src/main/assets/game.js
      - name: Build APK
        run: gradle --no-daemon :app:assembleDebug
      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: Gevork-Billiardinsky-v4.1-APK
          path: app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error
